```typescript
function SeasonSummary({ games, season, position }: SeasonSummaryProps) {
  const avg = calculateSeasonAverages(games, season);
  if (!avg) return <div className="text-muted-foreground text-sm">No data for {season}</div>;

  // Filter and sort games for this season (ascending for chart)
  const seasonGames = games
    .filter(g => g.season === season)
    .sort((a, b) => a.week - b.week);

  // Transform for chart (only need week + fantasyPoints)
  const chartData = seasonGames
    .filter(g => g.fantasyPoints !== undefined)
    .map(g => ({
      week: g.week,
      fantasyPoints: g.fantasyPoints ?? 0,
    }));

  return (
    <div className="space-y-4">
      {/* Season header */}
      <div className="flex items-center justify-between">
        <h4 className="font-semibold">{season} Season</h4>
        <span className="text-sm text-muted-foreground">{avg.games} games</span>
      </div>

      {/* Fantasy Points Trend Chart - NEW */}
      {chartData.length > 1 && (
        <div className="bg-muted/20 rounded-lg p-4">
          <p className="text-xs text-muted-foreground mb-2 uppercase tracking-wide">Fantasy Points Trend</p>
          <FantasyPointsChart data={chartData} />
        </div>
      )}

      {/* Existing season averages display */}
      <div className="bg-muted/30 rounded-lg p-4">
        {/* ... existing StatCell grid unchanged ... */}
      </div>

      {/* Existing game-by-game table */}
      <div className="max-h-64 overflow-y-auto">
        {/* ... existing GameRow list unchanged ... */}
      </div>
    </div>
  );
}
```

Key implementation notes:
- Only show chart if chartData has >1 data point (line needs at least 2 points)
- Chart appears between header and averages grid for visual hierarchy
- Light background (bg-muted/20) differentiates chart from stat boxes
- Label above chart explains what it shows

Do NOT modify:
- Existing SeasonSummary stat averages display (StatCell grid)
- Existing GameRow table (max-h-64 with overflow scroll)
- Existing PlayerHistory outer structure (tabs, loading states)
