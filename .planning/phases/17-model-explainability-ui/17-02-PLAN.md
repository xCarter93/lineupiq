interface FeatureContribution {
  feature: string;
  contribution: number;
  direction: "positive" | "negative";
  display_name: string;
}

interface ExplainabilityResponse {
  position: string;
  target: string;
  base_value: number;
  prediction: number;
  contributions: FeatureContribution[];
}

async function fetchExplanation(
  position: string,
  target: string,
  features: PredictionFeatures
): Promise<ExplainabilityResponse> {
  const response = await fetch(
    `${API_BASE}/explain/${position}/${target}`,
    {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(features),
    }
  );
  if (!response.ok) {
    throw new Error(`Explain API error: ${response.status}`);
  }
  return response.json();
}
```

function getPrimaryTarget(position: string): string {
  switch (position) {
    case "QB": return "passing_yards";
    case "RB": return "rushing_yards";
    case "WR":
    case "TE": return "receiving_yards";
    default: return "rushing_yards";
  }
}

function getTargetLabel(position: string): string {
  switch (position) {
    case "QB": return "Passing Yards";
    case "RB": return "Rushing Yards";
    case "WR":
    case "TE": return "Receiving Yards";
    default: return "Stats";
  }
}
```
