from pydantic import BaseModel

class PlayerRoster(BaseModel):
    """Player roster entry from nflreadpy."""
    player_id: str  # gsis_id
    name: str  # full_name
    position: str
    team: str
    jersey_number: int | None
    height: str | None  # e.g., "6-2"
    weight: int | None
    college: str | None
    years_exp: int | None
    headshot_url: str | None

class RosterResponse(BaseModel):
    """Full roster response."""
    season: int
    players: list[PlayerRoster]
    count: int

class WeeklyStats(BaseModel):
    """Single week of player stats."""
    season: int
    week: int
    opponent_team: str | None
    passing_yards: float | None
    passing_tds: float | None
    interceptions: float | None
    rushing_yards: float | None
    rushing_tds: float | None
    carries: float | None
    receiving_yards: float | None
    receiving_tds: float | None
    receptions: float | None
    fantasy_points: float | None

class PlayerHistoryResponse(BaseModel):
    """Player historical stats response."""
    player_id: str
    player_name: str
    position: str
    seasons: list[int]
    games: list[WeeklyStats]
    total_games: int
```

"""Roster and player history API routes."""
from fastapi import APIRouter, HTTPException, Query
import nflreadpy as nfl

from lineupiq.data.fetchers import fetch_rosters, fetch_player_history
from lineupiq.api.schemas.roster import RosterResponse, PlayerRoster, PlayerHistoryResponse, WeeklyStats

router = APIRouter()

@router.get("/roster", response_model=RosterResponse)
async def get_roster(
    season: int = Query(default=None, description="NFL season year (default: current)")
) -> RosterResponse:
    """Get current NFL roster for fantasy-relevant positions."""
    if season is None:
        season = nfl.get_current_season()

    df = fetch_rosters([season])

    players = [
        PlayerRoster(
            player_id=row["gsis_id"],
            name=row["full_name"],
            position=row["position"],
            team=row["team"],
            jersey_number=row.get("jersey_number"),
            height=row.get("height"),
            weight=row.get("weight"),
            college=row.get("college"),
            years_exp=row.get("years_exp"),
            headshot_url=row.get("headshot_url"),
        )
        for row in df.to_dicts()
    ]

    return RosterResponse(season=season, players=players, count=len(players))

@router.get("/player/{player_id}/history", response_model=PlayerHistoryResponse)
async def get_player_history(
    player_id: str,
    seasons: int = Query(default=3, ge=1, le=5, description="Number of seasons to fetch")
) -> PlayerHistoryResponse:
    """Get player's historical weekly stats for the last N seasons."""
    current = nfl.get_current_season()
    season_list = list(range(current - seasons + 1, current + 1))

    df = fetch_player_history(player_id, season_list)

    if df.is_empty():
        raise HTTPException(status_code=404, detail=f"No stats found for player {player_id}")

    # Get player info from first row
    first = df.row(0, named=True)

    games = [
        WeeklyStats(
            season=row["season"],
            week=row["week"],
            opponent_team=row.get("opponent_team"),
            passing_yards=row.get("passing_yards"),
            passing_tds=row.get("passing_tds"),
            interceptions=row.get("passing_interceptions"),
            rushing_yards=row.get("rushing_yards"),
            rushing_tds=row.get("rushing_tds"),
            carries=row.get("carries"),
            receiving_yards=row.get("receiving_yards"),
            receiving_tds=row.get("receiving_tds"),
            receptions=row.get("receptions"),
            fantasy_points=row.get("fantasy_points"),
        )
        for row in df.to_dicts()
    ]

    return PlayerHistoryResponse(
        player_id=player_id,
        player_name=first.get("player_display_name") or first.get("player_name"),
        position=first.get("position"),
        seasons=season_list,
        games=games,
        total_games=len(games),
    )
```
